cmake_minimum_required(VERSION 3.20)
project(cartoon_openvino_probe LANGUAGES CXX)
find_package(OpenVINO REQUIRED COMPONENTS Runtime)
add_executable(cartoon_openvino_probe probe.cpp)
target_compile_features(cartoon_openvino_probe PRIVATE cxx_std_17)
target_link_libraries(cartoon_openvino_probe PRIVATE openvino::runtime)
if(MSVC)
  target_compile_options(cartoon_openvino_probe PRIVATE /utf-8 /W4)
endif()
install(TARGETS cartoon_openvino_probe RUNTIME DESTINATION bin)
